async function loadResults() { const { data, error } = await supabaseClient .from("game_results") .select("round_number, game_time, result") .order("round_number", { ascending: true }); if (error) { console.error(error); return; } const table = document.getElementById("resultsTable"); table.innerHTML = ""; data.forEach(game => { const row = document.createElement("tr"); row.innerHTML = ` Game ${String(game.round_number).padStart(2, "0")} ${game.game_time} ${game.result || "--"} `; table.appendChild(row); }); } loadResults();
